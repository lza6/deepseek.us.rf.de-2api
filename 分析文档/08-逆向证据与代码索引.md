### 3.3 error / quota 解析（`handleErrorEvent`）

```js
let _ = null, p = null;
if (o.data) try {
    let l = JSON.parse(o.data);
    if (l.error) _ = l.error;
    if (l.quota_notice) p = l.quota_notice;
} catch {
    if (typeof o.data === "string" && o.data.length > 0 && o.data.length < 200) _ = o.data;
}
else if (o.target && o.target.readyState === EventSource.CLOSED) _ = "Connection was closed.";
else if (o.message) _ = o.message;
callback(p ? { type: "quota_notice", notice: p, message: _ || p.message || "" } : u, false, true);
```

### 3.5 访客 UUID

```js
function m() {  // generateGuestUuid
    return ("10000000-1000-4000-8000" + -1e11).replace(/[018]/g, e =>
        (e ^ crypto.getRandomValues(new Uint8Array(1))[0] & 15 >> e / 4).toString(16));
}
function o() {  // ensureGuestUuid
    let e = localStorage.getItem("aipkit_guest_uuid");
    return e || (e = m(), localStorage.setItem("aipkit_guest_uuid", e)),
        window.aipkit_guest_uuid = e, e;
}
```

## 5. 结论置信度分级

| 结论 | 置信度 | 依据 |
|------|--------|------|
| SSE 端点 URL 与参数 | **高** | 代码逐字确认 |
| SSE 事件类型（9 种） | **高** | addEventListener 确认 |
| delta 字段名 | **高** | `r.delta` 确认 |
| message_start 字段 message_id | **高** | 代码确认 |
| error/quota_notice 结构 | **高** | 代码确认 |
| dsgt REST 端点 | **高** | buy-tokens.js 确认 |
| 会话列表 action 名 | **中** | 间接（自研脚本调用） |
| 会话列表响应结构 | **低** | 推测（`conversations||items`） |
| provider 枚举 | **高** | 分支判断确认 |
| 具体模型 ID | **无** | 前端不含 |
| 免费额度数值 | **无** | 服务端 |
| 分包功能协议 | **无** | 未抓取 |
| Turnstile 触发点 | **低** | 推测 |
| sitekey 值 | **无** | 动态注入 |
